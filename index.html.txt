const Verify = {
  code: '', timer: null, secondsLeft: 300, isVerified: false, isLocked: false,

  start() {
    if (this.isVerified) return;
    this.stopTimer();
    this.secondsLeft = 300;
    this.code = String(Math.floor(100000 + Math.random() * 900000));
    this.isLocked = false;
    this.updateDisplay();
    this.updateTimerDisplay();
    this.setMessage('', '');
    document.getElementById('v-btn').disabled = false;
    document.getElementById('v-input').disabled = false;
    document.getElementById('v-input').value = '';
    document.getElementById('v-refresh').classList.remove('show');
    document.getElementById('v-timer').style.display = 'block';
    this.startTimer();
    setTimeout(() => document.getElementById('v-input')?.focus(), 100);
  },

  startTimer() {
    this.timer = setInterval(() => {
      this.secondsLeft--;
      this.updateTimerDisplay();
      if (this.secondsLeft <= 0) { this.stopTimer(); this.expire(); }
    }, 1000);
  },

  stopTimer() { if (this.timer) { clearInterval(this.timer); this.timer = null; } },

  updateTimerDisplay() {
    const el = document.getElementById('v-countdown');
    if (!el) return;
    const m = Math.floor(this.secondsLeft / 60);
    const s = this.secondsLeft % 60;
    el.textContent = String(m).padStart(2, '0') + ':' + String(s).padStart(2, '0');
  },

  updateDisplay() {
    const d = document.getElementById('code-display');
    if (d) { d.textContent = this.code; d.classList.remove('ok'); }
  },

  setMessage(text, type) {
    const m = document.getElementById('v-msg');
    if (!m) return;
    m.textContent = text;
    m.className = 'v-msg' + (type ? ' ' + type : '');
  },

  expire() {
    this.isLocked = true;
    this.setMessage('⏰ Код истёк. Нажмите «Обновить код».', 'error');
    document.getElementById('v-btn').disabled = true;
    document.getElementById('v-refresh').classList.add('show');
    document.getElementById('v-input').disabled = true;
  },

  copy() {
    if (!this.code) return;
    if (navigator.clipboard) {
      navigator.clipboard.writeText(this.code).then(() => {
        const d = document.getElementById('code-display');
        d.classList.add('ok');
        setTimeout(() => d.classList.remove('ok'), 1000);
      }).catch(() => {});
    }
  },

  check() {
    if (this.isLocked) { this.setMessage('⏰ Код истёк. Обновите код.', 'error'); return; }
    const input = document.getElementById('v-input');
    const val = input.value.trim();
    if (val.length !== 6) { this.setMessage('❌ Введите 6 цифр', 'error'); return; }
    if (val === this.code) { this.success(); }
    else {
      this.setMessage('❌ Неверный код. Попробуйте снова.', 'error');
      input.value = '';
      input.focus();
      const card = document.querySelector('.verify-card');
      if (card) { card.style.animation = 'none'; setTimeout(() => { card.style.animation = 'shake 0.4s'; }, 10); }
    }
  },

  success() {
    this.isVerified = true;
    this.stopTimer();
    this.setMessage('✅ Доступ разрешён!', 'success');
    document.getElementById('v-btn').disabled = true;
    document.getElementById('v-input').disabled = true;
    const d = document.getElementById('code-display');
    if (d) { d.textContent = '✓✓✓✓✓✓'; d.classList.add('ok'); }

    setTimeout(() => {
      const ov = document.getElementById('verify-overlay');
      ov.classList.add('hidden');
      setTimeout(() => {
        ov.style.display = 'none';
        try { bootApp(); } catch (e) { console.error('bootApp error:', e); }
      }, 500);
    }, 400);
  },

  restart() { this.start(); }
};

function initUser() {

  const tgUser = tg?.initDataUnsafe?.user;

  if (tgUser) {
    user = {
      id: 'tg_' + tgUser.id,
      telegramId: tgUser.id,
      firstName: tgUser.first_name || 'Пользователь',
      lastName: tgUser.last_name || '',
      username: tgUser.username || '',
      photoUrl: tgUser.photo_url || '',
      email: null
    };
    try { tg.ready(); tg.expand(); tg.setHeaderColor?.('#0a0e17'); tg.setBackgroundColor?.('#0a0e17'); } catch (_) {}
  } else {
    let id = localStorage.getItem('veloce_uid');
    if (!id) { id = 'web_' + Date.now(); localStorage.setItem('veloce_uid', id); }
    user = { id, firstName: 'Гость', username: '', email: null, telegramId: null };
  }

  const av = document.getElementById('avatarInitial');
  if (av) av.textContent = (user.firstName || '?')[0].toUpperCase();

  if (firebaseReady && db) {
    db.ref('users/' + user.id).once('value').then(snap => {
      const u = snap.val();
      if (u) {
        user.email = u.email || null;
        subscription = u.subscription || null;
      }
      db.ref('users/' + user.id).update({
        firstName: user.firstName,
        username: user.username,
        telegramId: user.telegramId,
        lastSeen: Date.now()
      }).catch(() => {});
      render();
    }).catch(() => render());
  }
}

function selectPayment(method) {
  if (method !== 'crypto') {
    toast('⏳ Этот способ оплаты скоро будет доступен', '⏳');
    return;
  }
  selectedPayment = method;
  document.querySelectorAll('.pay-method').forEach(m => {
    m.classList.toggle('active', m.dataset.pay === method && !m.classList.contains('disabled'));
  });
}

async function processPayment() {
  const btn = document.getElementById('payBtn');
  if (!selectedPlan) return;
  btn.disabled = true;
  btn.textContent = '⏳ Создаём счёт...';

  try {
    const orderId = 'VLC-' + Date.now().toString(36).toUpperCase().slice(-6);

    // ============================================================
    // СОЗДАНИЕ СЧЁТА В CRYPTOBOT
    // ============================================================
    // ВАЖНО: API-ключ CryptoBot нельзя держать в открытом коде —
    // его видно через DevTools. Поэтому запрос идёт через ваш Worker.
    //
    // Замените URL ниже на ваш Cloudflare Worker:
    // const WORKER_URL = 'https://velocepn-bot.ваш-субдомен.workers.dev';
    //
    // try {
    //   const res = await fetch(WORKER_URL + '/create-invoice', {
    //     method: 'POST',
    //     headers: { 'Content-Type': 'application/json' },
    //     body: JSON.stringify({
    //       userId: user.id,
    //       telegramId: user.telegramId,
    //       planId: selectedPlan.id,
    //       planName: selectedPlan.name,
    //       amount: selectedPlan.price,
    //       days: selectedPlan.days,
    //       devices: selectedPlan.devices,
    //       orderId
    //     })
    //   });
    //   const data = await res.json();
    //   if (data.pay_url) {
    //     window.open(data.pay_url, '_blank');
    //   } else {
    //     throw new Error(data.error || 'Ошибка CryptoBot');
    //   }
    // } catch (e) {
    //   throw new Error('Не удалось создать счёт: ' + e.message);
    // }

    // ============================================================
    // ДЕМО-РЕЖИМ (пока Worker не настроен)
    // ============================================================
    // Здесь мы просто записываем заказ в Firebase.
    // Когда вы поднимете Worker — раскомментируйте блок выше
    // и удалите строки ниже (кроме сохранения заказа).
    // ============================================================

    const order = {
      id: orderId,
      userId: user.id,
      userName: user.firstName,
      username: user.username,
      planId: selectedPlan.id,
      planName: selectedPlan.name,
      amount: selectedPlan.price,
      days: selectedPlan.days,
      devices: selectedPlan.devices,
      payment: 'crypto',
      status: 'pending',
      createdAt: Date.now(),
      telegramId: user.telegramId
    };

    if (firebaseReady && db) {
      await db.ref('orders/' + orderId).set(order);
    }

    // Уведомление админу
    if (CONFIG.botToken && CONFIG.adminChatId) {
      fetch(`https://api.telegram.org/bot${CONFIG.botToken}/sendMessage`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          chat_id: CONFIG.adminChatId,
          text: `🛒 Новый заказ #${orderId}\n\n` +
                `👤 ${user.firstName} ${user.username ? '@' + user.username : ''}\n` +
                `💼 ${selectedPlan.name} (${selectedPlan.period})\n` +
                `💰 ${selectedPlan.price} ₽\n` +
                `💳 CryptoBot\n` +
                `🆔 User: ${user.id}\n\n` +
                `⏳ Ожидает подтверждения оплаты`
        })
      }).catch(() => {});
    }

    // Демо: активируем подписку сразу (потом — только после webhook от CryptoBot)
    const now = Date.now();
    const sub = {
      planId: selectedPlan.id,
      startedAt: now,
      expiresAt: now + selectedPlan.days * 86400000,
      devices: selectedPlan.devices,
      configUrl: 'vless://demo-' + orderId,
      orderId
    };
    subscription = sub;
    if (firebaseReady && db) {
      await db.ref('users/' + user.id + '/subscription').set(sub);
      await db.ref('orders/' + orderId).update({ status: 'completed', completedAt: now });
    }

    closeOverlay('checkoutOverlay');
    showSuccess(orderId, selectedPlan);

  } catch (err) {
    console.error(err);
    toast('❌ ' + (err.message || 'Ошибка'), '❌');
    btn.disabled = false;
    btn.textContent = 'Попробовать снова';
  }
}

function openOverlay(id) {
  document.getElementById(id).classList.add('open');
  document.body.style.overflow = 'hidden';
}
function closeOverlay(id) {
  document.getElementById(id).classList.remove('open');
  document.body.style.overflow = '';
}

document.querySelectorAll('.overlay').forEach(o => {
  o.addEventListener('click', e => { if (e.target === o) closeOverlay(o.id); });
});

function toast(text, icon = '✅') {
  const t = document.getElementById('toast');
  document.getElementById('toastIcon').textContent = icon;
  document.getElementById('toastText').textContent = text;
  t.classList.add('show');
  clearTimeout(t._timer);
  t._timer = setTimeout(() => t.classList.remove('show'), 2200);
}

function copyText(text) {
  if (navigator.clipboard) navigator.clipboard.writeText(text).then(() => toast('Скопировано', '📋'));
}

function openSupport() {
  const tg = window.Telegram?.WebApp;
  const url = 'https://t.me/' + (CONFIG.supportUsername || '').replace('@', '');
  if (tg?.openTelegramLink) tg.openTelegramLink(url);
  else window.open(url, '_blank');
}

function updateBottomNav(show) {
  document.getElementById('bottomNav').classList.toggle('hidden', !show);
}
